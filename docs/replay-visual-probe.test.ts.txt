function deepCopy(state: GameState): GameState {
  const copy = JSON.parse(JSON.stringify(state));
  copy.tileStates = new Map();
  if (state.tileStates) state.tileStates.forEach((v: any, k: string) => copy.tileStates.set(k, { ...v, damagedEntities: v.damagedEntities ? new Set(v.damagedEntities) : undefined }));
  if (state.tilesBeingVacated) copy.tilesBeingVacated = new Set(state.tilesBeingVacated);
  return copy;
}
// verbatim from Game.tsx
function resetReplayProjectilesToTurnStart(copy: GameState) {
  if (!copy.activeProjectiles) return;
  const nowMs = Date.now();
  for (const proj of copy.activeProjectiles) {
    if (proj.tilePath && proj.tilePath.length > 0 && (proj as any)._turnStartTileIndex !== undefined) {
      const startIdx = Math.min((proj as any)._turnStartTileIndex, proj.tilePath.length - 1);
      proj.currentTileIndex = startIdx;
      const tileTransitMs = 800 / (proj.speed || 4);
      proj.tileEntryTime = nowMs - (startIdx * tileTransitMs);
      proj.startTime = nowMs - (startIdx * tileTransitMs);
    }
  }
}
// mirror of drawProjectile position + shrink logic (AnimatedGameBoard.tsx 5215-5330)
function drawn(p: Projectile, vsMap: Map<string, ProjectileVisualState>, now: number, frozen: boolean, curTurn: number) {
  if (!p.active) return null;
  const vs = vsMap.get(p.id);
  const vsFresh = !!vs && (vs.lastUpdateTurn === undefined || vs.lastUpdateTurn === curTurn);
  const useLogical = !vs || (frozen && !vsFresh);
  const x = useLogical ? p.logicalX : vs!.x;
  const y = useLogical ? p.logicalY : vs!.y;
  let scale = 1;
  let consumeAtMs: number | null = null;
  const tileTransitMs = p.isHoming && p.tilePath && p.tilePath.length > 1 ? 800 / (p.tilePath.length - 1) : 800 / (p.speed || 4);
  const anchorMs = p.tileEntryTime ?? p.startTime ?? now;
  if (p.despawning && p.despawnStartTime !== undefined) {
    const progress = Math.min(1, Math.max(0, (now - p.despawnStartTime) / TARGET_LOST_LINGER_MS));
    const f = 1 - progress ** 3; if (f <= 0.01) return null; scale *= f;
  } else if (p.hitResult?.deactivate && !p.hitResult.vfxSprite && !p.hitResult.deferredDeathEntityId && !p.hitResult.placeCollectibleConfig) {
    consumeAtMs = anchorMs + p.hitResult.hitTileIndex * tileTransitMs;
  } else if (!p.hitResult && p.tilePath && p.tilePath.length > 0) {
    let willFizzle = false;
    if (p.isHoming) {
      if (p.attackData.range !== undefined && p.pathTraveled !== undefined) {
        const remaining = p.attackData.range - p.pathTraveled;
        willFizzle = remaining < 0.5 || (p.homingPathStyle === 'pathfinding' && remaining < 1);
      }
    } else willFizzle = true;
    if (willFizzle) consumeAtMs = anchorMs + (p.tilePath.length - 1) * tileTransitMs;
  }
  if (consumeAtMs !== null) {
    const tuc = consumeAtMs - now;
    if (tuc <= DESPAWN_SHRINK_MS) {
      const progress = Math.min(1, Math.max(0, 1 - tuc / DESPAWN_SHRINK_MS));
      const f = 1 - progress ** 3; if (f <= 0.01) return null; scale *= f;
    }
  }
  return { x, y, scale, dir: p.direction };
}
